# 读写路径与刷新机制

## 1. fread 数据路径

### 1.1 入口

```c
// iofread.c:29-40
size_t _IO_fread (void *buf, size_t size, size_t count, FILE *fp)
{
  size_t bytes_requested = size * count;
  _IO_acquire_lock (fp);
  bytes_read = _IO_sgetn (fp, (char *) buf, bytes_requested);
  _IO_release_lock (fp);
  return bytes_requested == bytes_read ? count : bytes_read / size;
}
```

### 1.2 完整调用链

```
fread(buf, size, count, fp)
  └→ _IO_fread                              // iofread.c:30
       ├→ _IO_acquire_lock(fp)               // 获取流锁
       └→ _IO_sgetn(fp, buf, n)              // vtable __xsgetn
            └→ _IO_file_xsgetn(fp, buf, n)   // fileops.c:1349-1438
                 │
                 ├─ 缓冲区有数据 (read_ptr < read_end)?
                 │   └→ memcpy(用户buf, read_ptr, have) // 直接拷贝
                 │
                 ├─ 请求量 < 缓冲区大小?
                 │   └→ __underflow(fp)       // 填充缓冲区，再拷贝
                 │       └→ _IO_new_file_underflow  // fileops.c:511-588
                 │            └→ _IO_SYSREAD(fp, buf_base, buf_size)
                 │                └→ __read(fd, buf, n)  // 系统调用
                 │
                 └─ 请求量 >= 缓冲区大小?
                     └→ _IO_SYSREAD(fp, 用户buf, 对齐后的字节数)
                        // 大块数据直接读入用户缓冲区，绕过 FILE 缓冲
```

### 1.3 xsgetn 核心逻辑

`_IO_file_xsgetn()`（`fileops.c:1348-1438`）的三阶段策略：

**阶段 1：消费缓冲区现有数据**

```c
have = fp->_IO_read_end - fp->_IO_read_ptr;  // 缓冲区中可用字节数
if (want <= have) {
    memcpy (s, fp->_IO_read_ptr, want);       // 全部从缓冲区满足
    fp->_IO_read_ptr += want;
    want = 0;
}
```

**阶段 2：小请求 → 通过 underflow 重新填充缓冲区**

```c
if (want < (fp->_IO_buf_end - fp->_IO_buf_base)) {
    __underflow (fp);  // 从内核读数据填满缓冲区
    continue;          // 回到阶段 1 拷贝
}
```

**阶段 3：大请求 → 直接读到用户缓冲区**

```c
// 对齐到缓冲区块大小
count = want;
if (block_size >= 128)
    count -= want % block_size;        // 读整数个块
count = _IO_SYSREAD (fp, s, count);    // 直接 read() 到用户 buf
```

### 1.4 underflow：缓冲区填充

`_IO_new_file_underflow()`（`fileops.c:511-588`）：

```c
int _IO_new_file_underflow (FILE *fp)
{
  // EOF 粘滞性（C99 要求）
  if (fp->_flags & _IO_EOF_SEEN)
    return EOF;

  // 如果是行缓冲/无缓冲，先刷新 stdout
  if (fp->_flags & (_IO_LINE_BUF | _IO_UNBUFFERED)) {
    if ((stdout->_flags & (_IO_LINKED | _IO_NO_WRITES | _IO_LINE_BUF))
        == (_IO_LINKED | _IO_LINE_BUF))
      _IO_OVERFLOW (stdout, EOF);      // 刷新 stdout
  }

  // 重置所有指针，准备读入
  fp->_IO_read_base = fp->_IO_read_ptr = fp->_IO_buf_base;
  fp->_IO_read_end = fp->_IO_buf_base;
  fp->_IO_write_base = fp->_IO_write_ptr = fp->_IO_write_end = fp->_IO_buf_base;

  // 从内核读取
  count = _IO_SYSREAD (fp, fp->_IO_buf_base,
                       fp->_IO_buf_end - fp->_IO_buf_base);
  fp->_IO_read_end += count;
  return *(unsigned char *) fp->_IO_read_ptr;
}
```

**行缓冲刷新 stdout 的设计意图：** 当 stdin 是行缓冲或无缓冲时，在读取输入前
先刷新 stdout 的行缓冲数据，确保用户看到提示信息再输入。

## 2. fwrite 数据路径

### 2.1 入口

```c
// iofwrite.c:29-78
size_t _IO_fwrite (const void *buf, size_t size, size_t count, FILE *fp)
{
  _IO_acquire_lock (fp);
  if (_IO_vtable_offset (fp) != 0 || _IO_fwide (fp, -1) == -1)
    written = _IO_sputn (fp, (const char *) buf, request);
  _IO_release_lock (fp);
}
```

**注意：** `_IO_fwide(fp, -1)` 尝试将流设为字节方向。如果流已被设为
宽字符方向，则返回正值而非 -1，fwrite 不执行写入。

### 2.2 完整调用链

```
fwrite(buf, size, count, fp)
  └→ _IO_fwrite                             // iofwrite.c:30
       ├→ _IO_acquire_lock(fp)
       ├→ _IO_fwide(fp, -1)                 // 确保字节方向
       └→ _IO_sputn(fp, buf, n)             // vtable __xsputn
            └→ _IO_new_file_xsputn(fp,buf,n)// fileops.c:1274-1345
                 │
                 ├─ 行缓冲模式?
                 │   └→ 查找 '\n'，决定是否提前刷新
                 │
                 ├─ 缓冲区有空间?
                 │   └→ memcpy(write_ptr, 用户buf, count)  // 拷入缓冲区
                 │
                 ├─ 缓冲区满或需要刷新?
                 │   └→ _IO_OVERFLOW(fp, EOF)  // 刷新缓冲区到内核
                 │       └→ _IO_new_file_overflow // fileops.c:781-838
                 │            └→ _IO_do_write → new_do_write
                 │                └→ _IO_SYSWRITE(fp, buf, n)
                 │                    └→ __write(fd, buf, n) // 系统调用
                 │
                 └─ 大块剩余数据?
                     └→ new_do_write(fp, s, do_write)   // 直接写
                         └→ _IO_SYSWRITE(fp, s, count)
```

### 2.3 xsputn 核心逻辑

`_IO_new_file_xsputn()`（`fileops.c:1274-1345`）：

**行缓冲特殊处理（`1288-1303`）：**

```c
if ((f->_flags & _IO_LINE_BUF) && (f->_flags & _IO_CURRENTLY_PUTTING)) {
    count = f->_IO_buf_end - f->_IO_write_ptr;  // 可用空间
    if (count >= n) {
        // 数据全部能放入缓冲区，但查找最后一个 '\n'
        for (p = s + n; p > s; )
            if (*--p == '\n') {
                count = p - s + 1;   // 只拷到 '\n' 位置
                must_flush = 1;
                break;
            }
    }
}
```

**缓冲区满时刷新（`1317-1343`）：**

```c
if (to_do + must_flush > 0) {
    if (_IO_OVERFLOW (f, EOF) == EOF)  // 刷新现有缓冲区数据
        return ...;

    // 大块数据：对齐到缓冲区块大小，直接写
    block_size = f->_IO_buf_end - f->_IO_buf_base;
    do_write = to_do - (block_size >= 128 ? to_do % block_size : 0);
    if (do_write)
        count = new_do_write (f, s, do_write);  // → write() 系统调用

    // 剩余尾部数据拷入缓冲区
    if (to_do)
        _IO_default_xsputn (f, s + do_write, to_do);
}
```

### 2.4 overflow：缓冲区刷新到内核

`_IO_new_file_overflow()`（`fileops.c:781-838`）：

```c
int _IO_new_file_overflow (FILE *f, int ch)
{
  // 如果不在写模式，切换到写模式
  if ((f->_flags & _IO_CURRENTLY_PUTTING) == 0) {
      _IO_doallocbuf (f);    // 确保有缓冲区
      // 设置写指针，清空读指针
      f->_IO_write_ptr = f->_IO_read_ptr;
      f->_IO_write_base = f->_IO_write_ptr;
      f->_IO_write_end = f->_IO_buf_end;
      f->_IO_read_base = f->_IO_read_ptr = f->_IO_read_end;
      f->_flags |= _IO_CURRENTLY_PUTTING;

      // 行缓冲/无缓冲：限制 write_end
      if (f->_flags & (_IO_LINE_BUF | _IO_UNBUFFERED))
          f->_IO_write_end = f->_IO_write_ptr;
  }

  if (ch == EOF)
      // 刷新请求：写出 write_base 到 write_ptr 的全部数据
      return _IO_do_write (f, f->_IO_write_base,
                           f->_IO_write_ptr - f->_IO_write_base);

  if (f->_IO_write_ptr == f->_IO_buf_end)
      _IO_do_flush (f);        // 缓冲区真的满了

  *f->_IO_write_ptr++ = ch;   // 写入字符

  // 无缓冲或行缓冲遇到 '\n'：立即刷新
  if ((f->_flags & _IO_UNBUFFERED)
      || ((f->_flags & _IO_LINE_BUF) && ch == '\n'))
      _IO_do_write (f, f->_IO_write_base,
                    f->_IO_write_ptr - f->_IO_write_base);

  return (unsigned char) ch;
}
```

## 3. fflush 机制

### 3.1 单流刷新

```c
// iofflush.c:31-44
int _IO_fflush (FILE *fp)
{
  if (fp == NULL)
    return _IO_flush_all ();         // 刷新所有流
  else {
    _IO_acquire_lock (fp);
    result = _IO_SYNC (fp) ? EOF : 0;  // vtable __sync
    _IO_release_lock (fp);
    return result;
  }
}
```

`_IO_new_file_sync()`（`fileops.c:841-871`）的工作：

```c
int _IO_new_file_sync (FILE *fp)
{
  // 如果有待写数据，先刷新
  if (fp->_IO_write_ptr > fp->_IO_write_base)
    if (_IO_do_flush(fp)) return EOF;

  // 计算已预读但未消费的字节数
  delta = fp->_IO_read_ptr - fp->_IO_read_end;  // 负值 = 预读了多少

  // 调整文件偏移：回退预读的字节
  if (delta != 0) {
    new_pos = _IO_SYSSEEK (fp, delta, SEEK_CUR);
    if (new_pos != EOF)
      fp->_IO_read_end = fp->_IO_read_ptr;  // 清空缓冲区
  }
  fp->_offset = _IO_pos_BAD;
}
```

**关键点：** sync 不仅刷新写缓冲，还会通过 `lseek(fd, delta, SEEK_CUR)`
回退文件位置以补偿预读的数据。

### 3.2 全局刷新：fflush(NULL)

`_IO_flush_all()`（`genops.c:708-755`）遍历 `_IO_list_all`
链表，对每个有待写数据的流执行 `_IO_OVERFLOW(fp, EOF)` + `_IO_SYNC(fp)`：

```c
int _IO_flush_all (void)
{
  int result = 0;
  _IO_lock_lock (list_all_lock);

  for (fp = (FILE *) _IO_list_all; fp != NULL; fp = fp->_chain) {
    _IO_flockfile (fp);

    // 跳过正在 freopen/fclose 的流
    if ((fp->_flags2 & _IO_FLAGS2_NOCLOSE) == 0) {
      // 有待写数据 → 刷新
      if ((fp->_mode <= 0 && fp->_IO_write_ptr > fp->_IO_write_base)
          || (fp->_mode > 0
              && fp->_wide_data->_IO_write_ptr > fp->_wide_data->_IO_write_base))
        if (_IO_OVERFLOW (fp, EOF) == EOF)
          result = EOF;

      // 有预读数据 → 同步文件位置
      if (_IO_fileno (fp) >= 0
          && ((fp->_mode <= 0 && fp->_IO_read_ptr < fp->_IO_read_end)
              || (fp->_mode > 0 && ...)))
        if (_IO_SYNC (fp) != 0)
          result = EOF;
    }

    _IO_funlockfile (fp);
  }

  _IO_lock_unlock (list_all_lock);
  return result;
}
```

### 3.3 程序退出时的刷新

```
exit()
  → __run_exit_handlers()
    → _IO_cleanup()
      → _IO_flush_all()    // 刷新所有流的写缓冲
      → _IO_unbuffer_all() // 释放所有流的缓冲区
```

## 4. fseek 与缓冲区

### 4.1 seekoff 概述

`_IO_new_file_seekoff()`（`fileops.c:968-1125`）处理 `fseek` 操作：

```
fseek(fp, offset, whence)
  → _IO_seekoff(fp, offset, whence, mode)
    → _IO_new_file_seekoff              // fileops.c:969
         ├→ 如果 mode==0 → do_ftell()     // ftell 快速路径
         ├→ 刷新写缓冲区（切换到读模式）
         ├→ 计算目标位置
         ├→ 尝试在缓冲区内定位（避免系统调用）
         └→ _IO_SYSSEEK(fp, offset, whence) // lseek 系统调用
              └→ 清空缓冲区，设置新的读指针
```

### 4.2 缓冲区内优化

对于 SEEK_CUR 和 SEEK_SET，如果目标位置在当前缓冲区范围内，
可以直接调整 `_IO_read_ptr` 而无需系统调用：

```c
// fileops.c 内 seekoff 逻辑（简化）
if (目标在 [read_base, read_end) 范围内) {
    fp->_IO_read_ptr = fp->_IO_read_base + offset;
    // 无需 lseek!
}
```

### 4.3 seek 后的缓冲区状态

seek 完成后，缓冲区被重置：

```c
fp->_IO_read_end = fp->_IO_read_ptr = fp->_IO_buf_base;
// 下次读取会触发 underflow 重新填充
```

## 5. printf 格式化输出路径

### 5.1 调用链

```
printf(fmt, ...)
  → __printf(fmt, ...)
    → vfprintf(stdout, fmt, args)
      → __vfprintf_internal(fp, fmt, args, mode_flags)
         // stdio-common/vfprintf-internal.c
         ├→ 创建 struct __printf_buffer_to_file buf
         ├→ __printf_buffer_to_file_init(&buf, fp)
         │    // printf_buffer_to_file.c:107-113
         │    └→ __printf_buffer_to_file_switch(&buf)
         │         // 直接写入 fp 的缓冲区，或使用暂存区
         ├→ 格式化处理：解析 % 转换说明
         │    └→ 写入 buf（自动切换 fp buffer / staging area）
         └→ __printf_buffer_to_file_done(&buf)
              // printf_buffer_to_file.c:115-122
              └→ __printf_buffer_flush_to_file(&buf)
                   // printf_buffer_to_file.c:48-104
                   ├→ 如果使用暂存区 → _IO_sputn(fp, stage, count)
                   └→ 如果直接写入 fp 缓冲 → 更新 fp->_IO_write_ptr
```

### 5.2 printf buffer → FILE 桥接

`printf_buffer_to_file.c` 实现了 printf 缓冲区与 FILE 缓冲区的桥接：

**初始化（`printf_buffer_to_file.c:27-45`）：**

```c
void __printf_buffer_to_file_switch (struct __printf_buffer_to_file *buf)
{
  if (buf->fp->_IO_write_ptr < buf->fp->_IO_write_end)
    // fp 有可用缓冲区空间 → 直接写入 fp 的缓冲区
    buf->base.write_ptr = buf->fp->_IO_write_ptr;
  else
    // fp 无空间 → 使用暂存区（stage buffer）
    buf->base.write_ptr = buf->stage;
}
```

**刷新（`printf_buffer_to_file.c:48-104`）：**
- 暂存区中的数据通过 `_IO_sputn(fp, stage, count)` 写入 FILE
- 直接写入 fp 缓冲区的数据通过更新 `fp->_IO_write_ptr` 标记

## 6. 线程安全机制

### 6.1 _IO_lock_t 结构

```c
// sysdeps/nptl/stdio-lock.h:26
typedef struct { int lock; int cnt; void *owner; } _IO_lock_t;
```

| 字段 | 含义 |
|------|------|
| `lock` | futex 锁字 |
| `cnt` | 递归计数器 |
| `owner` | 持有者线程指针（THREAD_SELF） |

### 6.2 递归锁实现

**加锁（`stdio-lock.h:37-52`）：**

```c
#define _IO_lock_lock(_name)                                    \
  do {                                                          \
    void *__self = THREAD_SELF;                                 \
    if (SINGLE_THREAD_P && (_name).owner == NULL)               \
      /* 单线程快速路径：无需 futex */                           \
      { (_name).lock = 1; (_name).owner = __self; }             \
    else if ((_name).owner != __self)                           \
      /* 其他线程持有 → futex 等待 */                            \
      { lll_lock ((_name).lock, LLL_PRIVATE);                   \
        (_name).owner = __self; }                               \
    else                                                        \
      /* 同一线程重入 → 递增计数 */                               \
      ++(_name).cnt;                                            \
  } while (0)
```

**解锁（`stdio-lock.h:70-84`）：**

```c
#define _IO_lock_unlock(_name)                                  \
  do {                                                          \
    if (SINGLE_THREAD_P && (_name).cnt == 0)                    \
      /* 单线程快速路径 */                                       \
      { (_name).owner = NULL; (_name).lock = 0; }               \
    else if ((_name).cnt == 0)                                  \
      /* 最外层解锁 → futex 唤醒 */                              \
      { (_name).owner = NULL;                                   \
        lll_unlock ((_name).lock, LLL_PRIVATE); }               \
    else                                                        \
      /* 内层解锁 → 递减计数 */                                   \
      --(_name).cnt;                                            \
  } while (0)
```

### 6.3 POSIX flockfile 接口

```
flockfile(fp)   → _IO_flockfile  → _IO_lock_lock(fp->_lock)
funlockfile(fp) → _IO_funlockfile → _IO_lock_unlock(fp->_lock)
ftrylockfile(fp) → _IO_ftrylockfile → _IO_lock_trylock(fp->_lock)
```

用户可以通过 `flockfile` / `funlockfile` 包裹多个 stdio 操作，
保证整个序列的原子性。

### 6.4 _IO_acquire_lock / _IO_release_lock

`_IO_acquire_lock`（`stdio-lock.h:98-103`）利用 GCC `__attribute__((cleanup))`
实现 RAII 风格的自动解锁：

```c
#define _IO_acquire_lock(_fp)                                    \
  do {                                                           \
    FILE *_IO_acquire_lock_file                                  \
        __attribute__((cleanup (_IO_acquire_lock_fct))) = (_fp); \
    _IO_flockfile (_IO_acquire_lock_file);

#define _IO_release_lock(_fp) ; } while (0)
```

即使函数通过 longjmp 或异常离开，cleanup 函数也会自动释放锁。

## 7. 宽字符流

### 7.1 fwide：设置流方向

`_IO_fwide()`（`iofwide.c:46-106`）决定流的字节/宽字符方向：

```c
int _IO_fwide (FILE *fp, int mode)
{
  // 已确定方向或只是查询 → 返回当前方向
  if (fp->_mode != 0 || mode == 0)
    return fp->_mode;

  // 设置宽字符方向 (mode > 0)
  if (mode > 0) {
    // 初始化 gconv 转换器（LC_CTYPE locale）
    struct gconv_fcts fcts;
    __wcsmbs_clone_conv (&fcts);
    cc->__cd_in.step = fcts.towc;    // 字节 → 宽字符
    cc->__cd_out.step = fcts.tomb;   // 宽字符 → 字节

    // 切换到宽字符 vtable
    _IO_JUMPS_FILE_plus (fp) = fp->_wide_data->_wide_vtable;
  }

  fp->_mode = mode;  // +1 = 宽字符, -1 = 字节
  return mode;
}
```

### 7.2 宽字符缓冲区

宽字符流使用独立的 `_wide_data` 缓冲区（`struct _IO_wide_data`）：

```
FILE
├── _IO_read_ptr/end/base    ← 字节缓冲区（原始字节数据）
├── _IO_write_ptr/end/base
├── _wide_data
│   ├── _IO_read_ptr/end/base  ← 宽字符缓冲区（wchar_t 数据）
│   ├── _IO_write_ptr/end/base
│   ├── _IO_state            ← mbstate_t 转换状态
│   └── _wide_vtable         ← 宽字符操作 vtable
└── _mode                    ← +1=宽, -1=窄, 0=未定
```

宽字符读取流程：
```
fgetwc(fp) → 宽字符 vtable → _IO_wfile_underflow
  → 从字节缓冲区取原始数据
  → gconv 转换器：字节 → wchar_t
  → 存入宽字符缓冲区
  → 返回 wchar_t
```
